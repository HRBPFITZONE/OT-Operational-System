function steps(r){
  var spv=r.RequesterRole==='Supervisor Warehouse',st=r.Status,bad=st==='Rejected';
  var names=spv?['Supervisor','Manager Operational','Director','HRD Acknowledgement','Approved']:['Manager Operational','Director','HRD Acknowledgement','Approved'];
  var idx=spv?{Draft:0,Revision:0,'Pending Manager':1,'Pending Director':2,'Pending HRD':3,Approved:4}:{Draft:0,Revision:0,'Pending Director':1,'Pending HRD':2,Approved:3};
  var cur=idx[st];if(st==='Rejected')cur=r.MgrStatus==='Rejected'?(spv?1:0):(spv?2:1);if(st==='Cancelled')cur=-1;
  var sub=function(n){return n==='Manager Operational'?r.MgrStatus:n==='Director'?r.DirStatus:n==='HRD Acknowledgement'?r.HrdStatus:''};

function auditCsv(){
  var cols_=['Timestamp','User','Role','Activity','ReqNo','Remarks','Browser','Device','IP'],q=function(v){v=String(v==null?'':v);return /[",\n]/.test(v)?'"'+v.replace(/"/g,'""')+'"':v};
  var csv='\ufeff'+cols_.join(',')+'\n'+(S.auditRows||[]).map(function(r){return cols_.map(function(c){return q(r[c])}).join(',')}).join('\n');
  dl(btoa(unescape(encodeURIComponent(csv))),'audit_trail_'+new Date().toISOString().slice(0,10)+'.csv','text/csv');
}
